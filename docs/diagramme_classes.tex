% ── Styles ──────────────────────────────────────────────────
\tikzset{
    % Classe UML (3 compartiments)
    umlclass/.style={
        rectangle split,
        rectangle split parts=3,
        rectangle split part fill={blue!20, white, white},
        draw=black,
        thick,
        rounded corners=2pt,
        font=\small,
        text width=5.5cm,
        minimum width=5.8cm,
        inner sep=6pt,
        anchor=north,
    },
    % Énumération UML
    umlenum/.style={
        rectangle split,
        rectangle split parts=2,
        rectangle split part fill={yellow!20, white},
        draw=black,
        thick,
        rounded corners=2pt,
        font=\small,
        text width=3.2cm,
        minimum width=3.5cm,
        inner sep=6pt,
        anchor=north,
    },
    % Flèche association
    assoc/.style={
        -,
        thick,
    },
    % Flèche composition (losange plein)
    compo/.style={
        -{Diamond[fill=black, length=8pt, width=6pt]},
        thick,
    },
    % Flèche agrégation (losange vide)
    aggre/.style={
        -{Diamond[open, length=8pt, width=6pt]},
        thick,
    },
    % Héritage
    inherit/.style={
        ->,
        thick,
        >={Triangle[open, length=8pt, width=8pt]},
    },
    % Cardinalité
    card/.style={
        font=\scriptsize\bfseries,
        fill=white,
        inner sep=1pt,
    },
}
