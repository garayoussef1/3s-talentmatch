% ── Styles ──────────────────────────────────────────────────
\tikzset{
    % Acteur (bonhomme bâton)
    actor/.style={
        inner sep=0pt,
        minimum size=0pt,
    },
    % Cas d'utilisation (ellipse)
    usecase/.style={
        ellipse,
        draw=black,
        thick,
        fill=blue!8,
        text width=3.2cm,
        align=center,
        font=\small,
        minimum height=1.2cm,
    },
    % Système (rectangle englobant)
    system/.style={
        rectangle,
        draw=black!70,
        thick,
        rounded corners=5pt,
        fill=gray!5,
        inner sep=15pt,
    },
    % Flèche association
    assoc/.style={
        -,
        thick,
        draw=black!70,
    },
    % Flèche include
    incl/.style={
        ->,
        dashed,
        thick,
        draw=black!60,
        >={Stealth[length=6pt]},
    },
    % Flèche extend
    ext/.style={
        ->,
        dashed,
        thick,
        draw=green!50!black,
        >={Stealth[length=6pt]},
    },
    % Flèche héritage acteur
    inherit/.style={
        ->,
        thick,
        draw=black,
        >={Triangle[open, length=8pt, width=8pt]},
    },
}

% Macro pour dessiner un acteur (bonhomme bâton)
\newcommand{\stickman}[3]{% #1=name, #2=position, #3=label
    \node[actor] (#1) at #2 {};
    % Tête
    \draw[thick] ([yshift=12pt]#1.center) circle (5pt);
    % Corps
    \draw[thick] ([yshift=7pt]#1.center) -- ([yshift=-5pt]#1.center);
    % Bras
    \draw[thick] ([yshift=3pt, xshift=-7pt]#1.center) -- ([yshift=3pt, xshift=7pt]#1.center);
    % Jambes
    \draw[thick] ([yshift=-5pt]#1.center) -- ([yshift=-14pt, xshift=-6pt]#1.center);
    \draw[thick] ([yshift=-5pt]#1.center) -- ([yshift=-14pt, xshift=6pt]#1.center);
    % Label
    \node[below=16pt, font=\small\bfseries] at (#1.center) {#3};
}
