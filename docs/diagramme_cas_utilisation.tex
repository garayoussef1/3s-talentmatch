% Bonhomme bâton ─────────────────────────────────────────────
\newcommand{\bonhomme}[3]{%  #1=nom  #2=(x,y)  #3=étiquette
  \node (#1) at #2 {};
  \draw[thick] ([yshift=13pt]#1.center) circle (5.5pt);
  \draw[thick] ([yshift=7pt]#1.center) -- ([yshift=-7pt]#1.center);
  \draw[thick] ([yshift=3pt,xshift=-8pt]#1.center)
            -- ([yshift=3pt,xshift= 8pt]#1.center);
  \draw[thick] ([yshift=-7pt]#1.center)
            -- ([yshift=-17pt,xshift=-7pt]#1.center);
  \draw[thick] ([yshift=-7pt]#1.center)
            -- ([yshift=-17pt,xshift= 7pt]#1.center);
  \node[below=20pt, font=\small\bfseries] at (#1.center) {#3};
}
